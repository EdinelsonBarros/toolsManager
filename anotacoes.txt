  CREATE MATERIALIZED VIEW "C_ERGON"."GTO_VWM_ORGANOGRAMA_SETORES" ("SETOR", "NOMESETOR", "PAISETOR", "NOMEPAISETOR", "SETORES_PAI", "ORGANOGRAMA", "HIERARQUIA_SETORES", "HIERARQUIA_NUM")
  ORGANIZATION HEAP PCTFREE 10 PCTUSED 40 INITRANS 1 MAXTRANS 255 
 NOCOMPRESS LOGGING
  STORAGE(INITIAL 65536 NEXT 65536 MINEXTENTS 1 MAXEXTENTS 2147483645
  PCTINCREASE 0 FREELISTS 1 FREELIST GROUPS 1
  BUFFER_POOL DEFAULT FLASH_CACHE DEFAULT CELL_FLASH_CACHE DEFAULT)
  TABLESPACE "C_ERGON_TS" 
  BUILD IMMEDIATE
  USING INDEX 
  REFRESH COMPLETE ON DEMAND
  USING DEFAULT LOCAL ROLLBACK SEGMENT
  USING ENFORCED CONSTRAINTS DISABLE QUERY REWRITE
  AS WITH 
SETORES_ATIVOS AS (
    SELECT 
        S.SUBEMP_CODIGO,
        S.SETOR,
        S.NOMESETOR,
        S.PAISETOR,
        P.NOMESETOR AS NOMEPAISETOR,   -- JOIN em vez de subquery
        S.CIDADE
    FROM HSETOR_ S
    LEFT JOIN HSETOR_ P
        ON P.SETOR = S.PAISETOR
       AND P.DATAFIM IS NULL
       AND P.EXTINTO = 'N'
    WHERE S.DATAFIM IS NULL 
      AND S.EXTINTO = 'N'
),
ORGAN_SETORES AS (
    SELECT
        LPAD(' ', 2 * (LEVEL - 1)) || SA.SETOR AS organograma,
        SYS_CONNECT_BY_PATH(SA.SETOR, '|') AS SETORES_PAI,
        SA.*,
        LEVEL AS NIVEL
    FROM SETORES_ATIVOS SA
    START WITH SA.PAISETOR IS NULL
    CONNECT BY PRIOR SA.SETOR = SA.PAISETOR
),
HIERARQUIA (SETOR, NOMESETOR, PAISETOR, NIVEL, COD_ESTRUTURADO ) AS (
    SELECT 
        SA.SETOR,
        SA.NOMESETOR,
        SA.PAISETOR,
        SA.NIVEL,
        CAST(SA.SUBEMP_CODIGO AS VARCHAR2(500)) AS COD_ESTRUTURADO
    FROM ORGAN_SETORES SA
    WHERE SA.NIVEL = 2
    
    UNION ALL
    
    SELECT 
        T.SETOR, 
        T.NOMESETOR, 
        T.PAISETOR,
        T.NIVEL,
        H.COD_ESTRUTURADO || '.' || T.ORDEM_FILHO
    FROM (
        SELECT 
            SETOR, PAISETOR, NOMESETOR, NIVEL,
            ROW_NUMBER() OVER (PARTITION BY PAISETOR ORDER BY NOMESETOR) AS ORDEM_FILHO
        FROM ORGAN_SETORES SA
        WHERE NIVEL > 2
    ) T
    INNER JOIN HIERARQUIA H ON T.PAISETOR = H.SETOR
)
SELECT
    OS.SETOR,
    OS.NOMESETOR,
    OS.PAISETOR,
    OS.NOMEPAISETOR,
    OS.SETORES_PAI,
    OS.ORGANOGRAMA,
    H.COD_ESTRUTURADO AS HIERARQUIA_SETORES,
	TO_NUMBER(REPLACE(H.COD_ESTRUTURADO, '.', '')) AS HIERARQUIA_NUM
FROM SETORES_ATIVOS SA
INNER JOIN ORGAN_SETORES OS ON OS.SETOR = SA.SETOR
INNER JOIN HIERARQUIA H ON H.SETOR = SA.SETOR
;

  CREATE MATERIALIZED VIEW "C_ERGON"."GTO_VWM_ORGANOGRAMA_SERVIDORES" ("NUMFUNC", "NOME", "NOME_CARGO_FUNCAO", "TIPO_CARGO", "SETOR", "NOMESETOR", "PAISETOR", "NOMEPAISETOR", "SETORES_PAI", "ORGANOGRAMA", "HIERARQUIA_SETORES", "HIERARQUIA_NUM", "HIERARQUIA_SERVIDORES")
  ORGANIZATION HEAP PCTFREE 10 PCTUSED 40 INITRANS 1 MAXTRANS 255 
 NOCOMPRESS LOGGING
  STORAGE(INITIAL 65536 NEXT 65536 MINEXTENTS 1 MAXEXTENTS 2147483645
  PCTINCREASE 0 FREELISTS 1 FREELIST GROUPS 1
  BUFFER_POOL DEFAULT FLASH_CACHE DEFAULT CELL_FLASH_CACHE DEFAULT)
  TABLESPACE "C_ERGON_TS" 
  BUILD IMMEDIATE
  USING INDEX 
  REFRESH COMPLETE ON DEMAND
  USING DEFAULT LOCAL ROLLBACK SEGMENT
  USING ENFORCED CONSTRAINTS DISABLE QUERY REWRITE
  AS 
  WITH ORGAN AS (
    SELECT 
        OS.* -- <--- Usando sua coluna jÃ¡ preenchida
    FROM GTO_VWM_ORGANOGRAMA_SETORES OS
)
SELECT 
    
    FB.NUMFUNC,
    FB.NOME,
    CASE WHEN FB.CARGO IS NOT NULL THEN FB.FLEX_CAMPO_10 ELSE FB.FLEX_CAMPO_11 END AS NOME_CARGO_FUNCAO,
    C.FLEX_CAMPO_02 AS TIPO_CARGO,
    OG.*,
    -- Criamos o Ã­ndice final: INDICE_SETOR + 3 dÃ­gitos para o cargo/servidor
    -- Multiplicamos por 1000 para ""abrir"" as casas decimais das pessoas
    (OG.HIERARQUIA_NUM * 1000) + 
        ROW_NUMBER() OVER (
            PARTITION BY OG.SETOR 
            ORDER BY 
                CASE WHEN C.FLEX_CAMPO_02 = 'CHEFIA' THEN 1 ELSE 2 END, 
                FB.NOME ASC
        ) AS HIERARQUIA_SERVIDORES

FROM FITABANCO FB
INNER JOIN ORGAN OG 
    ON OG.SETOR = FB.SETOR
LEFT JOIN CARGOS_ C 
    ON C.CARGO = NVL(FB.CARGO, FB.FUNCAO)
WHERE FB.MES_ANO = TRUNC(SYSDATE, 'MM')  
  AND (FB.NUMERO = 1 OR FB.CATEGORIA = 'ESTAGIARIO')
  AND pack_ergon.get_situacao_func(FB.numfunc, FB.numvinc, NULL, SYSDATE) = 'ATIVO'
  AND FB.SETOR = '002.SUTRA'
;
